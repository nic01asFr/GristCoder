<script>
let tab=0,demandes=[],biens=[],marches=[],interventions=[],selId=null,selType='demande',bpuItems=[],bcMarcheId=null,bcLines=[];
const BC_TYPES=['Maintenance','Exploitation'];

function showTab(t,el){tab=t;document.querySelectorAll('.tab').forEach(b=>b.classList.remove('active'));el.classList.add('active');closePanel();render();}
function closePanel(){document.getElementById('panel').classList.remove('open');selId=null;}
function bien(id){return biens.find(b=>b.id===id)?.fields?.Nom||'—';}
function formatDate(ts){return ts?new Date(ts*1000).toLocaleDateString('fr-FR'):'—';}
function urgBadge(u){const c={Critique:'C',Haute:'H',Normale:'N'}[u]||'N';return`<span class="badge badge-${c}">${u||'Normale'}</span>`;}

function render(){
  const list=document.getElementById('list');
  const dm=demandes.filter(d=>d.fields.Orientation==='Maintenance'||d.fields.Statut==='En cours Maintenance'||d.fields.Orientation==='Exploitation');
  document.getElementById('k1').textContent=dm.filter(d=>!d.fields.Statut||d.fields.Statut==='Nouvelle'||d.fields.Statut==='Qualifiée').length;
  document.getElementById('k2').textContent=dm.filter(d=>d.fields.Statut==='En cours Maintenance').length;
  document.getElementById('k3').textContent=marches.filter(m=>m.fields.Statut==='En cours'&&BC_TYPES.includes(m.fields.Type_marche)).length;

  if(tab===0){
    if(!dm.length){list.innerHTML='<div class="empty"><div class="empty-icon">🔧</div><div>Aucune demande maintenance</div></div>';return;}
    list.innerHTML=dm.sort((a,b)=>{const o={Critique:0,Haute:1,Normale:2};return(o[a.fields.Urgence]||2)-(o[b.fields.Urgence]||2);}).map(d=>`
      <div class="row${selId===d.id&&selType==='demande'?' sel':''}" onclick="openDemande(${d.id})">
        <div class="dot dot-${(d.fields.Urgence||'N')[0]}"></div>
        <div class="row-body">
          <div class="row-top"><div class="row-titre">${d.fields.Description||'—'}</div>${urgBadge(d.fields.Urgence)}</div>
          <div class="row-meta">📍 ${bien(d.fields.Bien)} · ${d.fields.Localisation||''} · ${formatDate(d.fields.Date_saisie)}</div>
        </div>
      </div>`).join('');
  } else if(tab===1){
    if(!interventions.length){list.innerHTML='<div class="empty"><div class="empty-icon">🔧</div><div>Aucune intervention</div></div>';return;}
    list.innerHTML=interventions.map(i=>{
      const sc={Planifiée:'badge-prog','En cours':'badge-enc',Terminée:'badge-ok',Annulée:'badge-id'}[i.fields.Statut]||'badge-id';
      return`<div class="card">
        <div class="card-top"><div class="card-t">${i.fields.Description||'—'}</div><span class="badge ${sc}">${i.fields.Statut||'—'}</span></div>
        <div class="card-m">📍 ${bien(i.fields.Bien)} · 📅 ${formatDate(i.fields.Date_prevue)} · ${i.fields.Cout_ht?i.fields.Cout_ht.toLocaleString('fr-FR')+'€ HT':''}</div>
      </div>`;
    }).join('');
  } else {
    const myM=marches.filter(m=>BC_TYPES.includes(m.fields.Type_marche));
    if(!myM.length){list.innerHTML='<div class="empty"><div class="empty-icon">📑</div><div>Aucun marché maintenance</div></div>';return;}
    list.innerHTML=myM.map(m=>`
      <div class="card">
        <div class="card-top"><div class="card-t">${m.fields.Nom||'—'}</div><span class="badge badge-prog">${m.fields.Statut||'—'}</span></div>
        <div class="card-m">📑 ${m.fields.Ref_marche||''} · ${m.fields.Type_marche||''} · ${m.fields.Montant_ht?m.fields.Montant_ht.toLocaleString('fr-FR')+'€ HT':''}</div>
        <div class="card-m">📅 ${formatDate(m.fields.Date_notification)} → ${formatDate(m.fields.Date_fin)}</div>
      </div>`).join('');
  }
}

function openDemande(id){
  selId=id;selType='demande';
  const d=demandes.find(x=>x.id===id);
  if(!d)return;
  document.getElementById('ptitle').textContent='Demande #'+id;
  document.getElementById('panel-body').innerHTML=`
    <div class="pl">Bien</div><div class="pv">${bien(d.fields.Bien)}</div>
    <div class="pl">Localisation</div><div class="pv">${d.fields.Localisation||'—'}</div>
    <div class="pl">Description</div><div class="pv">${d.fields.Description||'—'}</div>
    <div class="pl">Urgence</div><div class="pv">${d.fields.Urgence||'—'}</div>
    <div class="pl">Type</div><div class="pv">${d.fields.Type_demande||'—'}</div>
    <div class="pl">Notes traitement</div>
    <textarea class="pinput" id="p-notes" rows="3">${d.fields.Notes_traitement||''}</textarea>`;
  document.getElementById('panel-footer').innerHTML=`
    <button class="abtn abtn-create" onclick="creerIntervention(${id})">🔧 Créer intervention</button>
    <button class="abtn abtn-close" onclick="cloturerDemande(${id})">✓ Clôturer</button>
    <button class="abtn" style="background:#dbeafe;color:#1d4ed8" onclick="openBCModal()">🧾 BC Marché</button>`;
  document.getElementById('panel').classList.add('open');render();
}

async function creerIntervention(demId){
  const d=demandes.find(x=>x.id===demId);
  if(!d)return;
  const notes=document.getElementById('p-notes')?.value||'';
  const desc=prompt('Description de l\'intervention :',d.fields.Description||'');
  if(!desc)return;
  try{
    await grist.docApi.applyUserActions([['BulkAddRecord','Interventions_GPI',[null],{
      Bien:[d.fields.Bien],Description:[desc],
      Date_prevue:[Math.floor(Date.now()/1000)],
      Statut:['Planifié'],Priorite:[d.fields.Urgence==='Critique'?'Urgente':d.fields.Urgence==='Haute'?'Haute':'Normale'],
      Demande:[demId]
    }]]);
    await grist.docApi.applyUserActions([['BulkUpdateRecord','Demandes',[demId],{Statut:['En cours Maintenance'],Notes_traitement:[notes]}]]);
    await refresh();closePanel();render();alert('✅ Intervention créée !');
  }catch(e){alert('Erreur: '+e.message);}
}

async function cloturerDemande(demId){
  const notes=document.getElementById('p-notes')?.value||'';
  try{
    await grist.docApi.applyUserActions([['BulkUpdateRecord','Demandes',[demId],{Statut:['Clôturée'],Notes_traitement:[notes],Date_cloture:[Math.floor(Date.now()/1000)]}]]);
    await refresh();closePanel();render();
  }catch(e){alert('Erreur: '+e.message);}
}

async function refresh(){
  const [d,i]=await Promise.all([grist.docApi.fetchTable('Demandes'),grist.docApi.fetchTable('Interventions_GPI')]);
  demandes=d.id.map((id,i)=>({id,fields:Object.fromEntries(Object.entries(d).map(([k,v])=>[k,v[i]]))}));
  interventions=i.id.map((id,j)=>({id,fields:Object.fromEntries(Object.entries(i).map(([k,v])=>[k,v[j]]))}));
}

async function init(){
  grist.ready({requiredAccess:'full'});
  const [d,b,m,i,bpu]=await Promise.all([
    grist.docApi.fetchTable('Demandes'),
    grist.docApi.fetchTable('Biens'),
    grist.docApi.fetchTable('Marches'),
    grist.docApi.fetchTable('Interventions_GPI'),
    grist.docApi.fetchTable('BPU_Marche')
  ]);
  biens=b.id.map((id,i)=>({id,fields:Object.fromEntries(Object.entries(b).map(([k,v])=>[k,v[i]]))}));
  demandes=d.id.map((id,i)=>({id,fields:Object.fromEntries(Object.entries(d).map(([k,v])=>[k,v[i]]))}));
  marches=m.id.map((id,i)=>({id,fields:Object.fromEntries(Object.entries(m).map(([k,v])=>[k,v[i]]))}));
  interventions=i.id.map((id,j)=>({id,fields:Object.fromEntries(Object.entries(i).map(([k,v])=>[k,v[j]]))}));
  bpuItems=bpu.id.map((id,j)=>({id,fields:Object.fromEntries(Object.entries(bpu).map(([k,v])=>[k,v[j]]))}));
  render();
}
init();

function openBCModal(){
  var mList=marches.filter(function(m){return BC_TYPES.includes(m.fields.Type_marche)&&m.fields.Statut==='En cours';});
  if(!mList.length){alert('Aucun marché actif pour ce métier');return;}
  bcMarcheId=mList[0].id;bcLines=[];
  renderBCModal();
  var obEl=document.getElementById('bcObjet');
  if(obEl&&selId){var d=demandes.find(function(x){return x.id===selId;});if(d&&d.fields.Description)obEl.value=d.fields.Description;}
  document.getElementById('bcModal').classList.add('open');
}
function renderBCModal(){
  var mList=marches.filter(function(m){return BC_TYPES.includes(m.fields.Type_marche)&&m.fields.Statut==='En cours';});
  var m=marches.find(function(x){return x.id===bcMarcheId;});
  document.getElementById('bcMarcheSelect').innerHTML=mList.map(function(x){
    return '<option value="'+x.id+'"'+(x.id===bcMarcheId?' selected':'')+'>'+(x.fields.Ref_marche||'')+' — '+(x.fields.Objet||x.fields.Nom||'')+'</option>';
  }).join('');
  document.getElementById('bcImputation').value=(m&&m.fields.Code_comptable)||'';
  var bpuList=bpuItems.filter(function(b){return b.fields.Marche===bcMarcheId&&b.fields.Actif!==false;});
  if(bpuList.length){
    document.getElementById('bcBPUBody').innerHTML=bpuList.map(function(b){
      var l=bcLines.find(function(x){return x.bpuId===b.id;});
      return '<tr><td><input type="checkbox" onchange="toggleBCLine('+b.id+',this.checked)"'+(l?' checked':'')+'></td>'
        +'<td>'+(b.fields.Code_article||'')+'</td><td>'+(b.fields.Designation||'')+'</td>'
        +'<td>'+(b.fields.Unite||'')+'</td><td>'+((b.fields.Prix_ht||0).toLocaleString('fr-FR'))+' €</td>'
        +'<td><input type="number" min="0" step="0.1" value="'+(l?l.qty:1)+'" style="width:62px;padding:3px 6px;border:1px solid #cbd5e1;border-radius:4px" onchange="updateBCQty('+b.id+',+this.value)"></td>'
        +'<td id="mht-'+b.id+'">'+(l?((b.fields.Prix_ht||0)*(l.qty||1)).toLocaleString('fr-FR')+'€':'')+'</td></tr>';
    }).join('');
  } else {
    document.getElementById('bcBPUBody').innerHTML='<tr><td colspan="7" style="text-align:center;color:#94a3b8;padding:10px">Forfait — saisir montant HT ci-dessous</td></tr>';
  }
  updateBCTotal();
}
function toggleBCLine(bpuId,checked){if(checked)bcLines.push({bpuId:bpuId,qty:1});else bcLines=bcLines.filter(function(l){return l.bpuId!==bpuId;});updateBCTotal();}
function updateBCQty(bpuId,qty){var l=bcLines.find(function(x){return x.bpuId===bpuId;});if(l)l.qty=qty;var b=bpuItems.find(function(x){return x.id===bpuId;});var el=document.getElementById('mht-'+bpuId);if(el&&b)el.textContent=((b.fields.Prix_ht||0)*qty).toLocaleString('fr-FR')+'€';updateBCTotal();}
function updateBCTotal(){var ht=bcLines.reduce(function(s,l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return s+(b?(b.fields.Prix_ht||0)*(l.qty||1):0);},0);var manual=+(document.getElementById('bcMontantManuel').value)||0;if(!bcLines.length)ht=manual;var ttc=ht*1.2;document.getElementById('bcTotal').innerHTML='<span>HT: <strong>'+ht.toLocaleString('fr-FR')+' €</strong></span> <span>TVA 20%: '+(ht*0.2).toLocaleString('fr-FR')+' €</span> <span style="color:#059669;font-size:14px">TTC: <strong>'+ttc.toLocaleString('fr-FR')+' €</strong></span>';}
function onBCMarcheChange(){bcMarcheId=+document.getElementById('bcMarcheSelect').value;bcLines=[];renderBCModal();}
async function createBC(){
  var objet=document.getElementById('bcObjet').value.trim();
  if(!objet){alert('Objet obligatoire');return;}
  var ht=bcLines.reduce(function(s,l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return s+(b?(b.fields.Prix_ht||0)*(l.qty||1):0);},0);
  var manual=+(document.getElementById('bcMontantManuel').value)||0;
  if(!bcLines.length)ht=manual;
  if(!ht){alert('Montant nul');return;}
  var ttc=Math.round(ht*1.2*100)/100;
  var imputation=document.getElementById('bcImputation').value;
  var numBC='BC-'+new Date().getFullYear()+'-'+String(Date.now()).slice(-5);
  try{
    var res=await grist.docApi.applyUserActions([['BulkAddRecord','BonsCommande',[null],{
      Num_bc:[numBC],Marche:[bcMarcheId],
      Demande:[selId||0],
      Date_emission:[Math.floor(new Date().setHours(0,0,0,0)/1000)],
      Objet_bc:[objet],Montant_ht:[ht],Taux_tva:[20],Montant_ttc:[ttc],
      Statut:['Emis'],Imputation_budgetaire:[imputation]
    }]]);
    var bcId=res.retValues&&res.retValues[0]&&res.retValues[0].rowIdsChanged&&res.retValues[0].rowIdsChanged[0];
    if(bcLines.length&&bcId){
      var N=bcLines.length;
      await grist.docApi.applyUserActions([['BulkAddRecord','Lignes_BC',new Array(N).fill(null),{
        BC:bcLines.map(function(){return bcId;}),
        BPU_ref:bcLines.map(function(l){return l.bpuId;}),
        Code_article:bcLines.map(function(l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return b?b.fields.Code_article||'':'';}),
        Designation:bcLines.map(function(l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return b?b.fields.Designation||'':'';}),
        Unite:bcLines.map(function(l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return b?b.fields.Unite||'':'';}),
        Quantite:bcLines.map(function(l){return l.qty||1;}),
        Prix_unitaire_ht:bcLines.map(function(l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return b?b.fields.Prix_ht||0:0;}),
        Montant_ht:bcLines.map(function(l){var b=bpuItems.find(function(x){return x.id===l.bpuId;});return b?(b.fields.Prix_ht||0)*(l.qty||1):0;}),
        Taux_tva:bcLines.map(function(){return 20;})
      }]]);
    }
    document.getElementById('bcModal').classList.remove('open');
    alert('BC '+numBC+' émis. TTC: '+ttc.toLocaleString('fr-FR')+' €');
  }catch(e){alert('Erreur: '+e.message);}
}
